│ RecordIO 二进制格式 (来自 dmlc-core/recordio.h)                                                                                                                                                      │
│                                                                                                                                                                                                      │
│ 每条记录: [magic:4B][lrecord:4B][data:NB][padding:0-3B]                                                                                                                                              │
│                                                                                                                                                                                                      │
│ magic = 0xced7230a (uint32)                                                                                                                                                                          │
│ lrecord 编码: cflag(3 bits) << 29 | length(29 bits)                                                                                                                                                  │
│   cflag == 0: 完整记录                                                                                                                                                                               │
│   cflag == 1/2/3: 多段记录（大记录拆分）                                                                                                                                                             │
│ padding: 对齐到 4 字节边界                                                                                                                                                                           │
│                                                                                                                                                                                                      │
│ IRHeader (数据的前 24 字节): struct 'IfQQ'                                                                                                                                                           │
│   flag: uint32 (0=单标签, >0=多标签数组长度)                                                                                                                                                         │
│   label: float32 (当 flag==0 时)                                                                                                                                                                     │
│   id: uint64                                                                                                                                                                                         │
│   id2: uint64                                                                                                                                                                                        │
│ 当 flag > 0 时，label 是后续 flag 个 float32 数组                                                                                                                                                    │
│                                                                                                                                                                                                      │
│ .idx 文件格式                                                                                                                                                                                        │
│                                                                                                                                                                                                      │
│ 文本文件，每行: index\toffset\n                                                                                                                                                                      │
│                                                                                                                                                                                                      │
│ 实现: dataset/recordio_reader.py (新文件)                                                                                                                                                            │
│                                                                                                                                                                                                      │
│ import struct                                                                                                                                                                                        │
│ import io                                                                                                                                                                                            │
│ from PIL import Image                                                                                                                                                                                │
│ import numpy as np                                                                                                                                                                                   │
│                                                                                                                                                                                                      │
│ kMagic = 0xced7230a                                                                                                                                                                                  │
│                                                                                                                                                                                                      │
│ class RecordIOReader:                                                                                                                                                                                │
│     """纯 Python MXNet RecordIO 读取器，不依赖 mxnet"""                                                                                                                                              │
│                                                                                                                                                                                                      │
│     def __init__(self, idx_path, rec_path):                                                                                                                                                          │
│         self.offsets = {}                                                                                                                                                                            │
│         with open(idx_path, 'r') as f:                                                                                                                                                               │
│             for line in f:                                                                                                                                                                           │
│                 idx, offset = line.strip().split('\t')                                                                                                                                               │
│                 self.offsets[int(idx)] = int(offset)                                                                                                                                                 │
│         self.rec_file = open(rec_path, 'rb')                                                                                                                                                         │
│                                                                                                                                                                                                      │
│     def read_idx(self, idx):                                                                                                                                                                         │
│         self.rec_file.seek(self.offsets[idx])                                                                                                                                                        │
│         # 读取 magic + lrecord                                                                                                                                                                       │
│         header_bytes = self.rec_file.read(8)                                                                                                                                                         │
│         magic, lrecord = struct.unpack('<II', header_bytes)                                                                                                                                          │
│         assert magic == kMagic                                                                                                                                                                       │
│         cflag = (lrecord >> 29) & 7                                                                                                                                                                  │
│         length = lrecord & ((1 << 29) - 1)                                                                                                                                                           │
│         # cflag > 0 表示多段记录（极少见，大文件才会）                                                                                                                                               │
│         data = self.rec_file.read(length)                                                                                                                                                            │
│         if cflag > 0:                                                                                                                                                                                │
│             # 继续读取后续段                                                                                                                                                                         │
│             while cflag != 3 and cflag != 0:                                                                                                                                                         │
│                 header_bytes = self.rec_file.read(8)                                                                                                                                                 │
│                 magic2, lrecord2 = struct.unpack('<II', header_bytes)                                                                                                                                │
│                 cflag = (lrecord2 >> 29) & 7                                                                                                                                                         │
│                 length2 = lrecord2 & ((1 << 29) - 1)                                                                                                                                                 │
│                 data += self.rec_file.read(length2)                                                                                                                                                  │
│         return data                                                                                                                                                                                  │
│                                                                                                                                                                                                      │
│     @staticmethod                                                                                                                                                                                    │
│     def unpack_header(data):                                                                                                                                                                         │
│         """解析 IRHeader: flag(I) + label(f) + id(Q) + id2(Q) = 24 bytes"""                                                                                                                          │
│         flag, label, id_, id2 = struct.unpack('<IfQQ', data[:24])                                                                                                                                    │
│         if flag == 0:                                                                                                                                                                                │
│             return label, data[24:]                                                                                                                                                                  │
│         else:                                                                                                                                                                                        │
│             # 多标签: flag 个 float32                                                                                                                                                                │
│             label_end = 24 + flag * 4                                                                                                                                                                │
│             labels = struct.unpack(f'<{flag}f', data[24:label_end])                                                                                                                                  │
│             return labels[0], data[label_end:]                                                                                                                                                       │
│                                                                                                                                                                                                      │
│     def close(self):                                                                                                                                                                                 │
│         self.rec_file.close()                                                                                                                                                                        │
